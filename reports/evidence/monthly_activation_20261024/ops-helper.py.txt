def need(ok,why):
    if not ok: raise ValueError(why)

def pin(path,expected):
    p=Path(path)
    need(p.is_absolute() and all(not x.is_symlink() for x in (p,*p.parents)), 'SYMLINK_OR_RELATIVE')
    st=p.stat();need(stat.S_ISREG(st.st_mode) and st.st_nlink==1,'FILE_CUSTODY')
    raw=p.read_bytes();need(hashlib.sha256(raw).hexdigest()==expected,'HASH_DRIFT:'+str(p));return raw

def create(path,raw,mode):
    p=Path(path);need(all(not x.is_symlink() for x in (p,*p.parents)),'CREATE_SYMLINK')
    fd=os.open(p,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,mode)
    try:
        with os.fdopen(fd,'wb') as f:f.write(raw);f.flush();os.fsync(f.fileno())
        directory=os.open(p.parent,os.O_RDONLY|os.O_DIRECTORY)
        try:os.fsync(directory)
        finally:os.close(directory)
    except BaseException:
        # Partial evidence/authority is never removed or overwritten by recovery.
        raise

def command(*args):
    return subprocess.run(['/usr/bin/systemctl',*args],check=True,capture_output=True,text=True,timeout=10,env={'PATH':'/usr/bin:/bin','LC_ALL':'C'}).stdout

def state(unit):
    raw=command('show',unit,'--no-pager','--property=ActiveState,SubState,UnitFileState,MainPID,ControlPID,Job,DropInPaths,NeedDaemonReload,FragmentPath')
    return dict(line.split('=',1) for line in raw.splitlines() if '=' in line)

def require_idle(row):
    need(all(row.get(k)==v for k,v in dict(ActiveState='inactive',SubState='dead',MainPID='0',ControlPID='0',Job='',DropInPaths='',NeedDaemonReload='no').items()),'SERVICE_OR_JOB_NOT_IDLE')

    need(d['positions']==[] and d['open_orders']==[],'NOT_FLAT')

    need(e['id']=='5d8a43a6-7aab-40e5-866f-ac5046813493' and e['status']=='filled' and e['side']=='sell' and e['symbol']=='NET' and Decimal(e['filled_qty'])==Decimal('0.325167779'),'NET_EXIT_NOT_TERMINAL')
    need(buy['id']=='705a1b5b-1d45-4ba1-97cf-9c646033d78d' and buy['status']=='filled','NET_ENTRY_CONFLICT')
    need(stop['status']=='canceled' and Decimal(stop['filled_qty'])==0 and d['native_ledger']=={},'LEGACY_NATIVE_NOT_TERMINAL')
    need(d['source_pins_match'] and d['coverage']['pagination_exhausted'] and d['broker_writes']==d['remote_writes']==0,'CAPTURE_NOT_READ_ONLY_OR_INCOMPLETE')
    need(0<=time.time_ns()//1000000-d['observed_ms']<5000,'BROKER_CAPTURE_STALE')
    account=d['account_after']
    need(Decimal(account['cash'])>=Decimal('400') and account['trading_blocked'] is False and account['account_blocked'] is False,'ACCOUNT_NOT_ADMISSIBLE')
    for k in ('pending_reg_taf_fees','accrued_fees'):need(Decimal(account[k])>=0,'ACCOUNT_LIABILITY_UNKNOWN')


def terminal_activity_ownership(d,seal,protocol):
    # Independent validation is necessary because the financial classifier can
    # reject pending liabilities before reaching its ownership checks.
    from research_lab.alpaca_monthly_cost_source_v1 import legacy_transition_orders,_ms
    exits=legacy_transition_orders(seal,account_id=d['account_after']['id'],before_ms=d['observed_ms'])
    orders={o['id']:o for o in [seal['entry'],*exits]}
    for actual in (d['entry_order'],d['exit_order']):
        expected=orders.get(actual['id']);need(expected is not None,'TERMINAL_OWNER_UNKNOWN')
        need(all(actual[k]==expected[k] for k in ('id','symbol','side','client_order_id','status')),'TERMINAL_ORDER_CONFLICT')
        need(_ms(actual['filled_at'])==_ms(expected['filled_at']),'TERMINAL_ORDER_FILL_TIME')
        need(all(Decimal(actual[k])==Decimal(expected[k]) for k in ('filled_qty','filled_avg_price')),'TERMINAL_ORDER_QUANTITY_PRICE')
    groups={k:[] for k in orders}
    for row in d['activities']:
        if row['activity_type']!='FILL':continue
        order=orders.get(row.get('order_id'));need(order is not None,'MONTHLY_COST_FOREIGN_FILL')
        need(row['side']==order['side'] and row['symbol']==order['symbol'] and _ms(row['transaction_time'])<=_ms(order['filled_at']),'TERMINAL_FILL_CONFLICT')
        need(_ms(seal['entry']['filled_at'])<=_ms(row['transaction_time']) if order['side']=='sell' else True,'TERMINAL_EXIT_BEFORE_ENTRY')
        groups[order['id']].append(row)
    for oid,rows in groups.items():
        order=orders[oid];qty=sum((Decimal(r['qty']) for r in rows),Decimal(0))
        need(qty==Decimal(order['filled_qty']) and qty>0,'TERMINAL_FILL_QUANTITY')
        average=sum((Decimal(r['qty'])*Decimal(r['price']) for r in rows),Decimal(0))/qty
        if average!=Decimal(order['filled_avg_price']):
            precision=protocol.get('execution_average_precision',{})
            places=precision.get('decimal_places')
            need(type(places) is int and 0<=places<=9 and precision.get('rounding')=='ROUND_HALF_UP' and precision.get('source_sha256') in {r.get('sha256') for r in protocol['sources'].values()},'TERMINAL_PRECISION_UNBOUND')
            need(average.quantize(Decimal(1).scaleb(-places),rounding=ROUND_HALF_UP)==Decimal(order['filled_avg_price']),'TERMINAL_FILL_PRICE')


def classify_costs(mode,calculate):
    try:
        costs=calculate()
        need(Decimal(costs['classification']['monthly_unposted_fee_upper_bound_usd'])==0,'MONTHLY_PENDING_UNEXPECTED')
        return dict(status='CLASSIFIED',classification=costs['classification'])
    except ValueError as exc:
        # Installation grants no entries and keeps the timer disabled. Preserve
        # the real unresolved broker liability; activation still fails closed.
        if mode!='install' or str(exc)!='MONTHLY_COST_LIABILITY_UNCLASSIFIED':raise
        return dict(status='COST_RECONCILIATION_PENDING',reason=str(exc),classification=None)
