 def test_install_pending_never_becomes_activation(self):
  def unknown():raise ValueError('MONTHLY_COST_LIABILITY_UNCLASSIFIED')
  self.assertEqual(ops.classify_costs('install',unknown),dict(status='COST_RECONCILIATION_PENDING',reason='MONTHLY_COST_LIABILITY_UNCLASSIFIED',classification=None))
  with self.assertRaisesRegex(ValueError,'LIABILITY_UNCLASSIFIED'):ops.classify_costs('activate',unknown)

 def test_actual_terminal_lineage_then_pending(self):
  import sys,copy
  root=Path(os.environ.get('MONTHLY_TEST_APP_ROOT',str(Path(__file__).resolve().parents[2])))
  if not (root/'research_lab').is_dir():self.skipTest('local saved actual capture only')
  sys.path.insert(0,str(root))
  d=json.loads(Path(__file__).with_name('cost-blocker-private.json').read_bytes())
  packet=Path(os.environ.get('MONTHLY_TEST_PACKET_ROOT',str(root/'.private/money_runner_completion_20261023/monthly-owner-packet-v2')))
  seal=json.loads((packet/'legacy-transition.json').read_bytes());protocol=json.loads((packet/'protocol.json').read_bytes())
  ops.terminal_activity_ownership(d,seal,protocol)
  foreign=copy.deepcopy(d);foreign['activities'].append({**next(r for r in d['activities'] if r['activity_type']=='FILL'),'order_id':'foreign'})
  def unknown():raise ValueError('MONTHLY_COST_LIABILITY_UNCLASSIFIED')
  with self.assertRaisesRegex(ValueError,'FOREIGN_FILL'):
   ops.terminal_activity_ownership(foreign,seal,protocol)
   ops.classify_costs('install',unknown)
  broken=copy.deepcopy(d);next(r for r in broken['activities'] if r['activity_type']=='FILL')['qty']='1'

 def test_no_rewrite_and_no_symlink(self):
  with tempfile.TemporaryDirectory() as root:
   p=Path(root).resolve()/'file';ops.create(p,b'first',0o600)
   with self.assertRaises(FileExistsError):ops.create(p,b'second',0o600)
   self.assertEqual(p.read_bytes(),b'first');self.assertEqual(p.stat().st_mode&0o777,0o600)
   link=p.with_name('link');link.symlink_to(p)
   with self.assertRaises(ValueError):ops.pin(link,hashlib.sha256(b'first').hexdigest())
   with self.assertRaises(ValueError):ops.create(link,b'bad',0o600)
